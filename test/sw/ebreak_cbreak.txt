/* Counts from 0 to 10.
** check for compressed ebreak instruction and
** modifying the mepc register to return to the correct address
*/



.file "sw.S"

.equ MIP.MSIP, 0x8
.equ MIE.MSIE, 0x8

.equ MIP.MTIP, 0x80
.equ MIE.MTIE, 0x80

.equ MIP.MEIP, 0x800
.equ MIE.MEIE, 0x800

.text
.global _boot

_boot:
    la     a0, handler
    csrw   mtvec, a0
    li	   a0, MIE.MTIE
    csrw   mie,a0
    csrrsi a0,mstatus,8
	addi   a0, zero, 10
    addi   a1, zero, 16
    sw     zero, 0(a1)
    
loop:
	lw   a2, 0(a1)
    addi a2, a2, 1
    sw   a2, 0(a1)

.option rvc
    ebreak
    c.lw   a3, 0(a1)
.option norvc

    beq  a3, a0, exit_loop
    j    loop

exit_loop:
	addi a4, zero, 1
    addi a5, zero, 0x3c
    sb   a4, 3(a5)
halt:
	j    halt

.balign 64

handler:
    csrr	t2, mcause
    addi    t4, zero, 3
    bne     t2, t4, not_ebreak /* exit if the exception is not ebreak */
    addi	t1, t1, 1
    li      t3, MIP.MTIP
    csrc	mip, t3
is_ebreak:
    csrr    t2, mtval
    andi    t3, t2, 1
    beqz    t3, compressed_ebreak
    csrr    t2, mepc
    addi    t2, t2, 4
    csrw    mepc, t2
    mret
compressed_ebreak:
    csrr    t2, mepc
    addi    t2, t2, 2
    csrw    mepc, t2
    mret
not_ebreak:
    mret
